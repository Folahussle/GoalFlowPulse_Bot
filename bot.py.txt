import os
from telegram.ext import Updater, CommandHandler
import requests
from dotenv import load_dotenv

# Load secrets from .env
load_dotenv()
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
FOOTBALL_API_TOKEN = os.getenv("FOOTBALL_API_TOKEN")

# Command: /start
def start(update, context):
    update.message.reply_text("Hello! GoalFlowPulse Bot is ready ⚽")

# Command: /match
def match(update, context):
    url = "https://api.football-data.org/v4/matches"
    headers = {"X-Auth-Token": FOOTBALL_API_TOKEN}
    response = requests.get(url, headers=headers)
    if response.status_code == 200:
        data = response.json()
        update.message.reply_text(str(data)[:500])  # show first 500 chars
    else:
        update.message.reply_text("Error fetching match data.")

# Run bot
updater = Updater(TELEGRAM_TOKEN, use_context=True)
dp = updater.dispatcher
dp.add_handler(CommandHandler("start", start))
dp.add_handler(CommandHandler("match", match))

updater.start_polling()
updater.idle()
